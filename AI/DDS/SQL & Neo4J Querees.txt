SQL - select famsize, avg(G3_x) as avg_final_grade FROM student_data GROUP BY famsize;
Neo4J - MATCH (s:Student_Performance)
	WITH s.absences_x AS Absences, AVG(s.G3_x) AS avg_final_grade
	RETURN Absences, avg_final_grade;

SQL - SELECT school, sex, age, failures_x, famsup_x FROM student_data WHERE failures_x > 0 AND famsup_x = 'no';
Neo4J - MATCH (s:Student_Performance)
	WHERE s.failures_x > 0 AND s.famsup_x = 'no'
	RETURN s.school AS school, s.sex AS sex, s.age AS age, s.failures_x AS failures, s.famsup_x AS famsup;

SQL - SELECT internet, activities_x FROM student_data WHERE internet = 'yes' AND activities_x = 'yes';
Neo4J - MATCH (s:Student_Performance)
	WHERE s.internet = 'yes' AND s.activities_x = 'yes'
	RETURN s.internet AS internet, s.activities_x AS activities;

SQL - SELECT guardian_x, COUNT(*) AS mother_count FROM student_data WHERE guardian_x = 'other' GROUP BY guardian_x;
Neo4J - MATCH (s:Parent_Info)
	WHERE s.guardian_x = 'other'
	RETURN s.guardian_x AS guardian, COUNT(*) AS other_count;

SQL - SELECT Pstatus, COUNT(*) AS student_count FROM student_data GROUP BY Pstatus;
Neo4J - MATCH (s:Student_Demographics)
	WITH s.Pstatus AS Pstatus, COUNT(*) AS student_count
	RETURN Pstatus, student_count;

SQL - SELECT school, sex, age, address FROM student_data WHERE address = 'U';
Neo4J - MATCH (s:Student_Demographics)
	WHERE s.address = 'U' 
	RETURN s.school AS school, s.sex AS sex, s.age AS age, s.address AS address;

SQL - SELECT address, COUNT(*) AS student_count FROM student_data GROUP BY address;
Neo4J - MATCH (s:Student_Demographic)
	WITH s.address AS address, COUNT(*) AS student_count
	RETURN address, student_count;
