db.Hourly.aggregate([
  { $group: { _id: { $dateToString: { format: "%Y-%m-%d", date: "$date" } }, avg_temperature: { $avg: "$temp" } } },
  { $sort: { _id: 1 } }
])

db.Temperature.aggregate([
  { $group: { _id: { $dateToString: { format: "%Y-%m-%d", date: "$date" } }, avg_temperature: { $avg: "$houtemp" } } },
  { $sort: { _id: 1 } }
])

db.Power.aggregate([
  { $group: { _id: { $dateToString: { format: "%Y-%m-%d", date: "$date" } }, avg_power: { $avg: "$elpower" } } },
  { $sort: { _id: 1 } }
])
